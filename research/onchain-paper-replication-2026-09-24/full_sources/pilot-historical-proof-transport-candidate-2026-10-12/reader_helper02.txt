class _SweepGitReader:
    """One bounded Git process for source proofs in one fresh claims sweep."""
    def __init__(self, root):
        self.root = Path(root).resolve()
        self.process = None
        self.selector = None
        self.closed = False
        self.cache = _source_body_cache(root)
        self.cache['reader'] = self

    def __enter__(self):
        return self

    def _start(self):
        import os
        import selectors
        if self.closed:
            raise ValueError('Git sweep reader already closed')
        if self.process is None:
            self.process = subprocess.Popen(['git','cat-file','--batch-command','--buffer'],
                cwd=self.root, env={**os.environ,'GIT_NO_LAZY_FETCH':'1'},
                stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,bufsize=0)
            self.selector = selectors.DefaultSelector()
            for stream,label in ((self.process.stdout,'output'),(self.process.stderr,'error')):
                os.set_blocking(stream.fileno(),False)
                self.selector.register(stream,selectors.EVENT_READ,label)
            os.set_blocking(self.process.stdin.fileno(),False)

    @staticmethod
    def _end(raw,count,headers):
        offset=0
        for _ in range(count):
            end=raw.find(b'\n',offset)
            if end<0:
                if len(raw)-offset>128:raise ValueError('Git batch header missing or oversized')
                return None
            if end-offset>=128:raise ValueError('Git batch header missing or oversized')
            line=raw[offset:end];offset=end+1
            if not headers:
                match=re.fullmatch(rb'[0-9a-f]{40} blob (0|[1-9][0-9]*)',line)
                if match:
                    size=int(match[1])
                    if size>_GIT_BODY:raise ValueError('Git blob body ceiling exceeded')
                    if len(raw)<offset+size+1:return None
                    if raw[offset+size:offset+size+1]!=b'\n':raise ValueError('Git batch short body or terminator')
                    offset+=size+1
        return offset

    def exchange(self, requests, *, headers=False):
        import os
        import selectors
        import time
        if type(requests) is not bytes or len(requests)>_GIT_REQUEST or not requests.endswith(b'\n'):
            raise ValueError('Git request ceiling exceeded')
        count=requests.count(b'\n')
        if not 0<count<=_GIT_PARTS:raise ValueError('Git request count exceeded')
        self._start()
        prefix=b'info ' if headers else b'contents '
        outgoing=b''.join(prefix+line+b'\n' for line in requests.splitlines())+b'flush\n'
        sent=0;output=bytearray();errors=bytearray();deadline=time.monotonic()+_GIT_SECONDS
        self.selector.register(self.process.stdin,selectors.EVENT_WRITE,'input')
        try:
            while True:
                left=deadline-time.monotonic()
                if left<=0:raise subprocess.TimeoutExpired('Git source sweep',_GIT_SECONDS)
                for key,_ in self.selector.select(min(left,.25)):
                    stream=key.fileobj
                    if key.data=='input':
                        try:n=os.write(stream.fileno(),outgoing[sent:sent+4096])
                        except BlockingIOError:continue
                        if n<=0:raise ValueError('Git input closed')
                        sent+=n
                        if sent==len(outgoing):self.selector.unregister(stream)
                    else:
                        try:part=os.read(stream.fileno(),65536)
                        except BlockingIOError:continue
                        if not part:raise ValueError('Git batch short body or terminator')
                        target=output if key.data=='output' else errors
                        limit=_GIT_BODY+128*_GIT_PARTS if key.data=='output' else 65536
                        if len(target)+len(part)>limit:raise ValueError('Git aggregate output ceiling exceeded')
                        target.extend(part)
                end=self._end(output,count,headers)
                if end is not None and sent==len(outgoing):
                    if end!=len(output):raise ValueError('Git batch trailing response')
                    if errors:raise ValueError('Git sweep diagnostics')
                    return bytes(output)
        finally:
            if self.selector is not None:
                try:self.selector.unregister(self.process.stdin)
                except KeyError:pass

    def __exit__(self, typ, primary, traceback):
        import os
        import time
        self.closed=True
        process=self.process
        def finish():
            if process is None:return
            if primary is not None:
                if process.poll() is None:process.kill()
                process.wait(timeout=5)
                return
            process.stdin.close()
            deadline=time.monotonic()+5
            while self.selector.get_map():
                if time.monotonic()>=deadline:raise subprocess.TimeoutExpired('Git sweep close',5)
                for key,_ in self.selector.select(.05):
                    try:part=os.read(key.fileobj.fileno(),65536)
                    except BlockingIOError:continue
                    if part:raise ValueError('Git batch trailing response')
                    self.selector.unregister(key.fileobj)
            if process.wait(timeout=5)!=0:raise ValueError('saved committed provenance cannot be verified')
        def stop():
            if process is not None and process.poll() is None:process.kill()
        try:
            # Preserve first ordinary consumer failure; still force retirement.
            if primary is None:finish()
            else:
                stop()
                if process is not None:process.wait(timeout=5)
        finally:
            import sys
            active=sys.exception() or primary
            self.cache['bodies'].clear();self.cache['bytes']=0
            _git_cleanup((stop,
                lambda:process.wait(timeout=5) if process is not None else None,
                lambda:process.stdin.close() if process is not None else None,
                lambda:process.stdout.close() if process is not None else None,
                lambda:process.stderr.close() if process is not None else None,
                lambda:self.selector.close() if self.selector is not None else None),active)
        return False
