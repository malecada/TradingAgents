def require(v,message):
 if not v:raise ValueError(message)
def encoded(value):
 def visit(x,depth=0):
  require(depth<=32,'detached metadata depth exceeded')
  if type(x) in (str,int,bool,type(None)):return
  if type(x)is float:require(math.isfinite(x),'finite metadata scalar');return
  if type(x)is list:
   require(len(x)<=32768,'metadata list bound');[visit(y,depth+1) for y in x];return
  if type(x)is dict:
   require(len(x)<=32768 and all(type(k)is str for k in x),'metadata map bound');[visit(y,depth+1) for y in x.values()];return
  raise ValueError('only detached primitive metadata; no producer objects or callbacks')
 visit(value);raw=json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode();require(len(raw)<=LIMIT,'detached metadata bound');return raw

_ISSUED=object()
class Authority:
 """Run weakref + primitive bytes only; never a terminal/owner closure."""
 __slots__=('_record','_run_ref')
 def __init__(self,record,run,*,_key):
  require(_key is _ISSUED and type(record)is Record,'genuine factory authority required');object.__setattr__(self,'_record',record);object.__setattr__(self,'_run_ref',weakref.ref(run))
 def __setattr__(self,k,v):raise AttributeError('immutable handoff authority')
 def lease(self):
  run=self._run_ref();require(run is not None,'original run no longer live');return check_metadata(self._record,run)
